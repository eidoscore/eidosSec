python# backend/app/models.py
class License(Base):
    __tablename__ = "licenses"
    
    id = Column(UUID, primary_key=True)
    key = Column(String(255), unique=True, nullable=False)  # e.g., "TEAM-ABC123-XYZ789"
    tier = Column(Enum("free", "team", "business", "enterprise"))
    max_users = Column(Integer)  # Free=1, Team=10, Business=50, Enterprise=null
    max_projects = Column(Integer)  # Free=3, others=null (unlimited)
    features = Column(JSONB)  # {"ai_enabled": true, "pr_creation": false, ...}
    expires_at = Column(TIMESTAMP)  # null = lifetime, else expiry date
    created_at = Column(TIMESTAMP, default=now())
    
# backend/app/services/license_service.py
class LicenseService:
    def validate_license(self, license_key: str) -> dict:
        """
        Validate license key and return feature flags.
        
        Returns:
        {
            "valid": true,
            "tier": "team",
            "max_users": 10,
            "features": {
                "ai_enabled": true,
                "pr_creation": false,
                "deep_scan": true,
                ...
            },
            "expires_at": "2025-12-31"
        }
        """
        license = db.query(License).filter_by(key=license_key).first()
        
        if not license:
            return {"valid": False, "error": "Invalid license key"}
        
        if license.expires_at and license.expires_at < datetime.now():
            return {"valid": False, "error": "License expired"}
        
        return {
            "valid": True,
            "tier": license.tier,
            "max_users": license.max_users,
            "features": license.features,
            "expires_at": license.expires_at
        }
    
    def check_feature(self, license_key: str, feature: str) -> bool:
        """
        Check if license has access to feature.
        
        Example:
        if not license_service.check_feature(license_key, "ai_enabled"):
            raise HTTPException(403, "Upgrade to Team for AI features")
        """
        license_data = self.validate_license(license_key)
        
        if not license_data["valid"]:
            return False
        
        return license_data["features"].get(feature, False)
